# Collapse Rate by Completed Over
ggplot(data %>% group_by(overs_completed) %>%
         summarise(collapse_rate = mean(collapse)), 
       aes(x = overs_completed, y = collapse_rate)) +
  geom_line(color = "orange") +
  geom_point(color = "orange") +
  labs(title = "Collapse Rate by Completed Over", x = "Completed Over", y = "Collapse Rate") +
  theme_minimal()

# Remove partnership_id and other non-numeric/categorical columns
cor_data <- data %>%
  select(where(is.numeric)) %>%
  select(-any_of(c("match_id", "Unnamed..0", "partnership_id"))) %>%
  na.omit()

# Correlation matrix
cor_matrix <- cor(cor_data, use = "complete.obs")

# Melt for ggplot
melted_cor <- melt(cor_matrix)

# Plot heatmap
ggplot(data = melted_cor, aes(x = Var1, y = Var2, fill = value)) +
  geom_tile(color = "white") +
  scale_fill_gradient2(low = "purple", high = "yellow", mid = "cyan", midpoint = 0,
                       limit = c(-1,1), space = "Lab", name="Correlation Coefficient") +
  theme_minimal() +
  theme(axis.text.x = element_text(angle = 45, vjust = 1, hjust = 1)) +
  coord_fixed() +
  labs(title = "Correlation Heatmap (No Partnership ID)", x = NULL, y = NULL)
